# Mazmorra Infinita: crear las 6 salas una vez; despues subir las regiones del mapa (server apagado).
mv create infinita normal --world-type FLAT --no-structures --no-adjust-spawn --generator-settings {"layers":[{"block":"minecraft:air","height":1}],"biome":"minecraft:the_void"}
mv modify infinita set difficulty hard
mv modify infinita set gamemode survival
mv modify infinita set pvp false
mv modify infinita set allow-weather false
mv gamerule set minecraft:spawn_mobs false infinita
mv gamerule set minecraft:keep_inventory true infinita
mv gamerule set minecraft:mob_griefing false infinita
mv gamerule set minecraft:advance_time false infinita
rg flag -w infinita __global__ build deny
rg flag -w infinita __global__ block-break deny
rg flag -w infinita __global__ block-place deny
rg flag -w infinita __global__ chest-access deny
rg flag -w infinita __global__ pvp deny
rg flag -w infinita __global__ mob-spawning allow
rg flag -w infinita __global__ interact allow
rg flag -w infinita __global__ use deny
execute in minecraft:infinita run setworldspawn 0 63 8
