# Fosa del Crustaceo (oceano_fosa.py). Con AstralOceano instalado y las regiones de la arena subidas:
mv import fosa normal -g AstralOceano:vacio --skip-folder-check
execute in minecraft:fosa run setworldspawn 4 112 93
execute in minecraft:fosa run gamerule minecraft:spawn_mobs false
execute in minecraft:fosa run gamerule minecraft:mob_griefing false
execute in minecraft:fosa run gamerule minecraft:advance_time false
execute in minecraft:fosa run gamerule minecraft:advance_weather false
execute in minecraft:fosa run gamerule minecraft:keep_inventory true
rg flag -w fosa __global__ block-break deny
rg flag -w fosa __global__ block-place deny
rg flag -w fosa __global__ pvp deny
